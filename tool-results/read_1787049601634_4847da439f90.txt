    86→/** Render an ROI% as a multiplier (499% → "5.0×"). */
    87→function roiAsMultiplier(roiPct: number | null | undefined): string {
    88→  if (roiPct == null || !Number.isFinite(roiPct)) return 'N/A';
    89→  return `${(roiPct / 100).toFixed(1)}×`;

   661→function ScenarioStat({
   662→  label,
   663→  value,
   664→  expectedValue,
   665→  format,
   666→  showDelta,
   667→}: {
   668→  label: string;
   669→  value: number | null;
   670→  expectedValue: number | null;
   671→  format: 'currency' | 'multiplier' | 'payback';
   672→  showDelta?: boolean;
   673→}) {
   674→  // Delta vs. expected — only rendered when showDelta (active != 'expected') AND both
   675→  // values are non-null.
   676→  const delta =
   677→    showDelta && value != null && expectedValue != null
   678→      ? formatDelta(value, expectedValue, format)
   679→      : null;
   680→  return (
   681→    <div className="mkt-card-quiet p-5">
   682→      <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-ink-muted">{label}</p>
   683→      <p className="mt-3 font-mono tnum text-[clamp(1.75rem,3vw,2.25rem)] font-bold tracking-[-0.02em] text-ink">
   684→        {format === 'currency' ? (
   685→          <CountUp
   686→            value={value ?? 0}
   687→            prefix="$"
   688→            decimals={0}
   689→            retriggerOnValueChange
   690→          />
   691→        ) : format === 'multiplier' ? (
   692→          value == null ? (
   693→            <span>N/A</span>
   694→          ) : (
   695→            <CountUp
   696→              value={value}
   697→              suffix="×"
   698→              decimals={1}
   699→              retriggerOnValueChange
   700→            />
   701→          )
   702→        ) : value == null ? (
   703→          <span>Never</span>
   704→        ) : (
   705→          <CountUp
   706→            value={value}
   707→            suffix=" months"
   708→            decimals={1}
   709→            retriggerOnValueChange
   710→          />
   711→        )}
   712→      </p>
   713→      {delta && (
   714→        <p className="mt-2 font-mono tnum text-[11px] text-ink-faint">{delta}</p>
   715→      )}
   716→    </div>
   717→  );
   718→}
   719→
   720→/** Format a scenario delta vs. expected: "+$58k vs expected" / "−3.9×" / "+5.5 mo". */
   721→function formatDelta(
   722→  value: number,
   723→  expectedValue: number,
   724→  format: 'currency' | 'multiplier' | 'payback'
   725→): string {
   726→  const diff = value - expectedValue;
   727→  const sign = diff > 0 ? '+' : diff < 0 ? '\u2212' : '';
   728→  const abs = Math.abs(diff);
   729→  if (format === 'currency') {
   730→    // Compact k tier for deltas — "$58k" not "$58,000".
   731→    const v =
   732→      abs >= 1000
   733→        ? `${(abs / 1000).toFixed(abs >= 10000 ? 0 : 1)}k`
   734→        : `${abs.toFixed(0)}`;
   735→    return `${sign}$${v} vs expected`;
   736→  }
   737→  if (format === 'multiplier') {
   738→    return `${sign}${abs.toFixed(1)}× vs expected`;
   739→  }
   740→  // payback
   741→  return `${sign}${abs.toFixed(1)} mo vs expected`;
   742→}
