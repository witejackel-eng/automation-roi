   575→// Small helper: derive the DecisionKey from a scenario result, so the table's
   576→// last row uses the canonical DECISION_LABELS instead of a hardcoded string.
   577→function scenarioDecisionKey(r: ScenarioResult): DecisionKey {

   583→function ScenarioTable({
   584→  results,
   585→  activeScenario,
   586→  onSelect,
   587→}: {
   588→  results: Record<ScenarioName, ScenarioResult>;
   589→  activeScenario: ScenarioName;
   590→  onSelect: (s: ScenarioName) => void;

   599→    ['Additional customers', (r) => r.additionalCustomers.toLocaleString('en-US', { maximumFractionDigits: 0 })],

   603→    ['Net annual benefit', (r) => formatCurrency(r.netAnnualBenefit)],
   604→    ['ROI', (r) => formatRoi(r.roiPct)],
   605→    ['Payback', (r) => formatPayback(r.paybackMonths)],
   606→    ['Viableo Decision', (r) => DECISION_LABELS[scenarioDecisionKey(r)]],

   613→          Scenario detail
   614→        </h3>
   615→        <p className="text-[12px] text-ink-muted">
   616→          Costs held constant across all scenarios; only benefit assumptions vary.
   617→        </p>
   618→      </div>
   619→      <div className="overflow-x-auto">
   620→        <table className="w-full text-[13px]">
   621→          <thead>
   622→            <tr className="border-b border-border bg-surface">
   623→              <th className="px-5 py-2.5 text-left font-medium text-ink-muted">Metric</th>
   624→              {SCENARIO_ORDER.map((s) => (
   625→                <th
   626→                  key={s}
   627→                  className={`px-5 py-2.5 text-right font-medium ${
   628→                    s === activeScenario ? 'text-ink' : 'text-ink-muted'
   629→                  }`}
   630→                >
   631→                  <button
   632→                    type="button"
   633→                    onClick={() => onSelect(s)}
   634→                    className="min-h-[44px] hover:text-ink"
   635→                  >
   636→                    {SCENARIO_LABELS[s]}
   637→                  </button>
   638→                </th>
   639→              ))}
   640→            </tr>
   641→          </thead>
   642→          <tbody>
   643→            {rows.map((r, i) => (
   644→              <tr key={i} className="border-b border-border last:border-b-0 hover:bg-surface/50">
   645→                <td className="px-5 py-2.5 text-ink">{r[0]}</td>
   646→                {SCENARIO_ORDER.map((s) => {
   647→                  const val = r[1](results[s]);
   648→                  const isActive = s === activeScenario;
   649→                  return (
   650→                    <td
   651→                      key={s}
   652→                      className={`px-5 py-2.5 text-right font-mono tnum ${
   653→                        isActive ? 'text-ink' : 'text-ink-muted'
   654→                      }`}
   655→                    >
   656→                      {val}
   657→                    </td>
   658→                  );
   659→                })}
   660→              </tr>
   661→            ))}
   662→          </tbody>
   663→        </table>
   664→      </div>
   665→    </div>
   666→  );
   667→}

   705→/** Floating scroll-to-top button — appears after 400px scroll. */
   706→function ScrollToTop() {
   707→  const [visible, setVisible] = React.useState(false);
   708→
   709→  React.useEffect(() => {
   710→    const onScroll = () => setVisible(window.scrollY > 400);
   711→    onScroll();
   712→    window.addEventListener('scroll', onScroll, { passive: true });
   713→    return () => window.removeEventListener('scroll', onScroll);
   714→  }, []);
   715→
   716→  if (!visible) return null;
   717→
   718→  return (
   719→    <button
   720→      type="button"
   721→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   722→      className="fixed bottom-20 right-4 z-20 flex size-10 items-center justify-center rounded-full border border-border bg-surface shadow-floating text-ink-muted hover:text-ink transition-colors duration-hover md:bottom-6"
   723→      aria-label="Back to top"
   724→    >
   725→      <ArrowUp className="size-4" strokeWidth={1.75} aria-hidden="true" />
   726→    </button>
   727→  );
   728→}
