    68→export function ResultsView() {
    69→  const {
    70→    inputs,
    71→    results,
    72→    recommendation,
    73→    entitlement,
    74→    go,
    75→    activeScenario,
    76→    setActiveScenario,
    77→  } = useApp(
    78→    useShallow((s) => ({
    79→      inputs: s.inputs,
    80→      results: s.results,
    81→      recommendation: s.recommendation,
    82→      entitlement: s.entitlement,
    83→      go: s.go,
    84→      activeScenario: s.activeScenario,
    85→      setActiveScenario: s.setActiveScenario,
    86→    }))
    87→  );
    88→
    89→  const { toast } = useToast();
    90→  const [saving, setSaving] = React.useState(false);
    91→  const [reportLoading, setReportLoading] = React.useState(false);
    92→  const [proposalLoading, setProposalLoading] = React.useState(false);
    93→  const [shareLoading, setShareLoading] = React.useState(false);
    94→  // Mirror the store's savedProjectId locally so the existing handlers keep
    95→  // working unchanged, but ALSO sync it back to the store on save so that
    96→  // the reopen flow (which sets the store's savedProjectId before navigating
    97→  // here) is respected — re-saving after a reopen updates the same project
    98→  // rather than creating a duplicate.
    99→  const storeSavedProjectId = useApp((s) => s.savedProjectId);
   100→  const [savedProjectId, setSavedProjectId] = React.useState<string | null>(
   101→    storeSavedProjectId
   102→  );
   103→  // If the store's savedProjectId changes (e.g. user reopened a project),
   104→  // sync the local state.
   105→  React.useEffect(() => {
   106→    setSavedProjectId(storeSavedProjectId);
   107→  }, [storeSavedProjectId]);
   108→
   109→  if (!inputs || !results || !recommendation) {
   110→    return (
   111→      <div className="mx-auto w-full max-w-[1200px] px-4 py-16 md:px-6">
   112→        <EmptyResults onBack={() => go('calculator')} />
   113→      </div>
   114→    );
   115→  }
   116→
   117→  // Active scenario drives every displayed figure. The store's
   118→  // `recommendation` (Expected) is preserved for the saved-project record.
   119→  const active: ScenarioResult = results[activeScenario];
   120→  const activeRec = recommend(active);
   121→  const activeDecision = activeRec.recommendation as DecisionKey;
   122→  const ent = entitlement;
   123→  const canSave = !!ent && has(ent, 'save_project');
   124→  const canReport = !!ent && has(ent, 'client_report');
   125→  const canProposal = !!ent && has(ent, 'proposal');
   126→  const canShare = !!ent && has(ent, 'share_links');
   127→
   128→  // ROI as a multiplier (499% → "5.0×"). Null when totalFirstYearCost = 0.
   129→  const roiMultiple =
   130→    active.roiPct == null ? null : active.roiPct / 100;
   131→
   132→  const handleSave = async () => {
   133→    if (!canSave) {
   134→      go('pricing');
   135→      return;
   136→    }
   137→    setSaving(true);
   138→    try {
   139→      const res = await fetch('/api/projects', {
   140→        method: 'POST',
   141→        headers: { 'Content-Type': 'application/json' },
   142→        body: JSON.stringify({
   143→          inputs,
   144→          results,
   145→          recommendation,
   146→          clientName: inputs.clientName,
   147→        }),
   148→      });
   149→      if (res.status === 403) {
   150→        toast({ title: 'Saving projects requires Pro.', variant: 'destructive' });
   151→        go('pricing');
   152→        return;
   153→      }
   154→      if (!res.ok) {
   155→        toast({ title: 'Could not save the project.', variant: 'destructive' });
   156→        return;
   157→      }
   158→      const data = (await res.json()) as { id: string };
   159→      setSavedProjectId(data.id);
   160→      toast({ title: 'Project saved.' });
   161→    } catch {
   162→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   163→    } finally {
   164→      setSaving(false);
   165→    }
   166→  };
   167→
   168→  const handleReport = async () => {
   169→    if (!canReport) {
   170→      go('pricing');
   171→      return;
   172→    }
   173→    if (!savedProjectId) {
   174→      await handleSave();
   175→    }
   176→    const projectId = savedProjectId;
   177→    if (!projectId) {
   178→      toast({ title: 'Save the project first.', description: 'Then generate the report.' });
   179→      return;
   180→    }
   181→    setReportLoading(true);
   182→    try {
   183→      const res = await fetch(`/api/projects/${projectId}/report`, {
   184→        method: 'POST',
   185→        headers: { 'Content-Type': 'application/json' },
   186→        body: JSON.stringify({}),
   187→      });
   188→      if (res.status === 403) {
   189→        toast({ title: 'Client report requires Pro.', variant: 'destructive' });
   190→        go('pricing');
   191→        return;
   192→      }
   193→      if (!res.ok) {
   194→        toast({ title: 'Could not generate the report.', variant: 'destructive' });
   195→        return;
   196→      }
   197→      const data = (await res.json()) as { pdfUrl: string };
   198→      toast({ title: 'Report generated.' });
   199→      window.open(data.pdfUrl, '_blank', 'noopener,noreferrer');
   200→    } catch {
   201→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   202→    } finally {
   203→      setReportLoading(false);
   204→    }
   205→  };
   206→
   207→  const handleProposal = async () => {
   208→    if (!canProposal) {
   209→      go('pricing');
   210→      return;
   211→    }
   212→    if (!savedProjectId) {
   213→      await handleSave();
   214→    }
   215→    const projectId = savedProjectId;
   216→    if (!projectId) {
   217→      toast({ title: 'Save the project first.', description: 'Then generate the proposal.' });
   218→      return;
   219→    }
   220→    setProposalLoading(true);
   221→    try {
   222→      const res = await fetch(`/api/projects/${projectId}/proposal`, {
   223→        method: 'POST',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({}),
   226→      });
   227→      if (res.status === 403) {
   228→        toast({ title: 'Proposal generation requires Pro.', variant: 'destructive' });
   229→        go('pricing');
   230→        return;
   231→      }
   232→      if (!res.ok) {
   233→        toast({ title: 'Could not generate the proposal.', variant: 'destructive' });
   234→        return;
   235→      }
   236→      const data = (await res.json()) as { pdfUrl: string };
   237→      toast({ title: 'Proposal generated.' });
   238→      window.open(data.pdfUrl, '_blank', 'noopener,noreferrer');
   239→    } catch {
   240→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   241→    } finally {
   242→      setProposalLoading(false);
   243→    }
   244→  };
   245→
   246→  const handleShare = async () => {
   247→    if (!canShare) {
   248→      go('pricing');
   249→      return;
   250→    }
   251→    if (!savedProjectId) {
   252→      await handleSave();
   253→    }
   254→    const projectId = savedProjectId;
   255→    if (!projectId) {
   256→      toast({ title: 'Save the project first.', description: 'Then create a share link.' });
   257→      return;
   258→    }
   259→    setShareLoading(true);
   260→    try {
   261→      const res = await fetch(`/api/projects/${projectId}/share`, {
   262→        method: 'POST',
   263→        headers: { 'Content-Type': 'application/json' },
   264→        body: JSON.stringify({}),
   265→      });
   266→      if (res.status === 403) {
   267→        toast({ title: 'Share links require Agency Pro.', variant: 'destructive' });
   268→        go('pricing');
   269→        return;
   270→      }
   271→      if (!res.ok) {
   272→        toast({ title: 'Could not create the share link.', variant: 'destructive' });
   273→        return;
   274→      }
   275→      const data = (await res.json()) as { shareId: string; url: string };
   276→      // Copy to clipboard + use Web Share API where supported (Master Spec §46).
   277→      const fullUrl = `${window.location.origin}${data.url}`;
   278→      if (navigator.share) {
   279→        try {
   280→          await navigator.share({
   281→            title: 'Viableo — Shared analysis',
   282→            text: 'A shared automation business case.',
   283→            url: fullUrl,
   284→          });
   285→        } catch {
   286→          /* user dismissed — still copy below */
   287→        }
   288→      }
   289→      try {
   290→        await navigator.clipboard.writeText(fullUrl);
   291→        toast({
   292→          title: 'Link copied.',
   293→          description: 'Share it with your client.',
   294→        });
   295→      } catch {
   296→        toast({
   297→          title: 'Share link created.',
   298→          description: fullUrl,
   299→        });
   300→      }
   301→    } catch {
   302→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   303→    } finally {
   304→      setShareLoading(false);
   305→    }
   306→  };
   307→
   308→  return (
   309→    <div className="mx-auto w-full max-w-[1200px] px-4 py-8 md:px-6 md:py-12">
   310→      <ScrollToTop />
   311→      {/* Back + prepared-for row */}
   312→      <div className="mb-6 flex items-center justify-between">
   313→        <Button
   314→          variant="ghost"
   315→          size="sm"
   316→          onClick={() => go('calculator')}
   317→          className="min-h-[44px] gap-1.5 text-ink-muted hover:text-ink"
   318→        >
   319→          <ArrowLeft className="size-4" strokeWidth={1.75} aria-hidden="true" />
   320→          Edit assumptions
   321→        </Button>
   322→        <p className="text-[13px] text-ink-muted">
   323→          Prepared for <span className="font-medium text-ink">{inputs.clientName}</span>
   324→        </p>
   325→      </div>
   326→
   327→      {/* Page header — references TERM.analysis + TERM.decision */}
   328→      <header className="mb-8">
   329→        <h1 className="font-display text-[32px] font-bold leading-[1.02] tracking-[-0.02em] text-ink md:text-[40px]">
   330→          {TERM.analysis}
   331→        </h1>
   332→        <p className="mt-2 max-w-[640px] text-[15px] leading-[1.55] text-ink-muted">
   333→          The {TERM.decision} and the figures that justify it. Switch scenarios to see how the
   334→          numbers hold up under conservative and upside assumptions.
   335→        </p>
   336→      </header>
   337→
   338→      {/* Hero summary card — ScenarioSlider + DecisionBadge + 3 headline figures */}
   339→      <section
   340→        aria-label="Viableo Decision summary"
   341→        className="mb-10 rounded-lg border border-border bg-surface-raised p-5 md:p-6"
   342→      >
   343→        <div className="mb-6 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   344→          <ScenarioSlider value={activeScenario} onChange={setActiveScenario} />
   345→          {/* Cross-fade on decision change via key + reveal-on-enter animation. */}
   346→          <div
   347→            key={activeDecision}
   348→            className="reveal-on-enter flex items-center gap-2"
   349→          >
   350→            <span className="text-[11px] uppercase tracking-wide text-ink-muted">
   351→              {TERM.decision}
   352→            </span>
   353→            <DecisionBadge decision={activeDecision} size="lg" animate />
   354→          </div>
   355→        </div>
   356→
   357→        <AnimatePresence mode="wait">
   358→          <motion.div
   359→            key={activeScenario}
   360→            initial={{ opacity: 0.6, y: 4 }}
   361→            animate={{ opacity: 1, y: 0 }}
   362→            exit={{ opacity: 0.6, y: -4 }}
   363→            transition={{ duration: 0.28, ease: [0.16, 1, 0.3, 1] }}
   364→            className="grid grid-cols-1 gap-6 sm:grid-cols-3 sm:gap-8"
   365→          >
   366→          {/* Annual Opportunity ($) */}
   367→          <HeroFigure
   368→            label="Annual opportunity"
   369→            sublabel={
   370→              active.isRevenueOpportunityOnly
   371→                ? 'Revenue opportunity, net of first-year cost'
   372→                : 'Net annual benefit, net of first-year cost'
   373→            }
   374→          >
   375→            <CountUp
   376→              value={active.netAnnualBenefit}
   377→              prefix="$"
   378→              decimals={0}
   379→              retriggerOnValueChange
   380→              className="text-[40px] font-medium leading-none text-ink"
   381→            />
   382→          </HeroFigure>
   383→
   384→          {/* ROI (×) */}
   385→          <HeroFigure label="ROI" sublabel="Return on first-year cost">
   386→            {roiMultiple == null ? (
   387→              <span className="font-mono tnum text-[40px] font-medium leading-none text-ink-muted">
   388→                N/A
   389→              </span>
   390→            ) : (
   391→              <CountUp
   392→                value={roiMultiple}
   393→                suffix="×"
   394→                decimals={1}
   395→                retriggerOnValueChange
   396→                className="text-[40px] font-medium leading-none text-ink"
   397→              />
   398→            )}
   399→          </HeroFigure>
   400→
   401→          {/* Payback (months) */}
   402→          <HeroFigure label="Payback" sublabel="Months to recoup first-year cost">
   403→            {active.paybackMonths == null ? (
   404→              <span className="font-mono tnum text-[40px] font-medium leading-none text-ink-muted">
   405→                Never
   406→              </span>
   407→            ) : (
   408→              <CountUp
   409→                value={active.paybackMonths}
   410→                suffix=" mo"
   411→                decimals={active.paybackMonths >= 10 ? 0 : 1}
   412→                retriggerOnValueChange
   413→                className="text-[40px] font-medium leading-none text-ink"
   414→              />
   415→            )}
   416→          </HeroFigure>
   417→        </motion.div>
   418→        </AnimatePresence>
   419→      </section>
   420→
   421→      {/* Verdict Stamp — the detailed sign-off (kept per Task 4-b). */}
   422→      <div className="mb-8 flex justify-center md:justify-start">
   423→        <VerdictStamp
   424→          recommendation={activeRec.recommendation}
   425→          paybackMonths={active.paybackMonths}
   426→          roiPct={active.roiPct}
   427→          netAnnualBenefit={active.netAnnualBenefit}
   428→          size="lg"
   429→        />
   430→      </div>
   431→
   432→      {/* Why this recommendation? (Section 6.9) — expandable bullet rationale,
   433→          each reason backed by a real number from the active scenario. */}
   434→      <WhyRecommendation
   435→        inputs={inputs}
   436→        results={results}
   437→        recommendation={activeRec}
   438→        className="mb-8"
   439→      />
   440→
   441→      {/* Recommendation copy (active scenario) */}
   442→      <p className="mb-10 max-w-[760px] text-[15px] leading-[1.55] text-ink-muted">
   443→        {activeRec.copy}
   444→      </p>
   445→
   446→      {/* Secondary KPI row */}
   447→      <div className="mb-10 grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   448→        <KpiCard
   449→          label="Annual labor savings"
   450→          value={formatCurrency(active.annualLaborSavings)}
   451→          variant="secondary"
   452→        />
   453→        <KpiCard
   454→          label={
   455→            active.isRevenueOpportunityOnly
   456→              ? 'Additional revenue opportunity'
   457→              : 'Additional gross profit'
   458→          }
   459→          value={formatCurrency(active.additionalGrossProfit)}
   460→          variant="secondary"
   461→        />
   462→        <KpiCard
   463→          label="First-year cost"
   464→          value={formatCurrency(active.totalFirstYearCost)}
   465→          variant="secondary"
   466→        />
   467→      </div>
   468→
   469→      {/* Stress test + break-even thresholds (Master Spec §29, §30, §31).
   470→          Placed after the key financials and before the exhibits, per the
   471→          §33 results-page section order: decision → financials → why →
   472→          confidence → stress test → break-even → ROI bridge → comparison →
   473→          assumptions → actions. */}
   474→      <StressTestSection inputs={inputs} activeScenario={activeScenario} />
   475→
   476→      {/* Breaking point slider (Phase 3.1) — the product's most distinctive moment.
   477→          Integrates after the stress test section. */}
   478→      <BreakingPointSlider inputs={inputs} activeScenario={activeScenario} />
   479→
   480→      {/* Confidence, explained (Phase 3.2) — below the verdict. */}
   481→      <ConfidenceExplainedSection inputs={inputs} />
   482→
   483→      {/* Verdict reveal with count-up (Phase 3.3) */}
   484→      <VerdictReveal
   485→        confidenceScore={computeConfidenceScore({}).score}
   486→        roiMultiple={roiMultiple}
   487→        netAnnualBenefit={active.netAnnualBenefit}
   488→        className="mb-10"
   489→      />
   490→
   491→      {/* Recurring economics view (Phase 3.4) — first-class monthly view */}
   492→      <RecurringEconomicsView
   493→        result={active}
   494→        monthlyAiApiCost={inputs.monthlyAiApiCost}
   495→        monthlySoftwareCost={inputs.monthlySoftwareCost}
   496→        platformApiCost={inputs.platformApiCost}
   497→        className="mb-10"
   498→      />
   499→
   500→      {/* Top drivers callout (Phase 2.4) */}
   501→      <TopDriversCallout inputs={inputs} activeScenario={activeScenario} />
   502→
   503→      {/* Exhibit 1 — ROI Bridge (active scenario) */}
   504→      <React.Suspense
   505→        fallback={
   506→          <ChartSkeleton title="Exhibit 1 — Labor savings drive the first-year benefit" />
   507→        }
   508→      >
   509→        <div className="mb-10">
   510→          <RoiBridge
   511→            result={active}
   512→            isRevenueOpportunityOnly={active.isRevenueOpportunityOnly}
   513→          />
   514→        </div>
   515→      </React.Suspense>
   516→
   517→      {/* Exhibit 2 — Scenario Comparison (all three) */}
   518→      <React.Suspense
   519→        fallback={
   520→          <ChartSkeleton title="Exhibit 2 — Expected scenario carries the strongest benefit" />
   521→        }
   522→      >
   523→        <div className="mb-10">
   524→          <ScenarioComparison
   525→            results={results}
   526→            isRevenueOpportunityOnly={active.isRevenueOpportunityOnly}
   527→          />
   528→        </div>
   529→      </React.Suspense>
   530→
   531→      {/* Scenario detail table (numeric, for the audit trail) */}
   532→      <ScenarioTable
   533→        results={results}
   534→        activeScenario={activeScenario}
   535→        onSelect={setActiveScenario}
   536→      />
   537→
   538→      {/* Assumptions */}
   539→      <div className="mb-10 mt-12">
   540→        <h2 className="mb-4 font-display text-[22px] font-semibold tracking-[-0.02em] text-ink">
   541→          Assumptions
   542→        </h2>
   543→        <p className="mb-4 text-[14px] text-ink-muted">
   544→          Every input that feeds the calculation. Fields tagged &quot;context&quot; do not
   545→          affect the dollar math.
   546→        </p>
   547→        <AssumptionsTable inputs={inputs} />
   548→      </div>
   549→
   550→      {/* Sticky secondary actions */}
   551→      <div className="sticky bottom-0 -mx-4 mt-8 border-t border-border bg-canvas/95 px-4 py-4 backdrop-blur supports-[backdrop-filter]:bg-canvas/80 md:static md:mx-0 md:bg-transparent md:px-0 md:py-0 md:backdrop-blur-none">
   552→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-end">
   553→          <EntitlementButton
   554→            allowed={canSave}
   555→            requiredTierLabel="Pro"
   556→            onClick={handleSave}
   557→            loading={saving}
   558→            variant="secondary"
   559→            icon={<Save className="size-4" strokeWidth={1.75} aria-hidden="true" />}
   560→          >
   561→            Save project
   562→          </EntitlementButton>
   563→          <EntitlementButton
   564→            allowed={canReport}
   565→            requiredTierLabel="Pro"

   748→/** Floating scroll-to-top button — appears after 400px scroll. */
   749→function ScrollToTop() {
   750→  const [visible, setVisible] = React.useState(false);
   751→
   752→  React.useEffect(() => {
   753→    const onScroll = () => setVisible(window.scrollY > 400);
   754→    onScroll();
   755→    window.addEventListener('scroll', onScroll, { passive: true });
   756→    return () => window.removeEventListener('scroll', onScroll);
   757→  }, []);
   758→
   759→  if (!visible) return null;
   760→
   761→  return (
   762→    <button
   763→      type="button"
   764→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   765→      className="fixed bottom-20 right-4 z-20 flex size-10 items-center justify-center rounded-full border border-border bg-surface shadow-floating text-ink-muted hover:text-ink transition-colors duration-hover md:bottom-6"
   766→      aria-label="Back to top"
   767→    >
   768→      <ArrowUp className="size-4" strokeWidth={1.75} aria-hidden="true" />
   769→    </button>
   770→  );
   771→}
