   100→/** Render an ROI% as a multiplier (499% → "5.0×"). */

   102→  if (roiPct == null || !Number.isFinite(roiPct)) return 'N/A';
   103→  return `${(roiPct / 100).toFixed(1)}×`;

   714→function ScenarioStat({
   715→  label,
   716→  value,
   717→  expectedValue,
   718→  format,
   719→  showDelta,
   720→}: {
   721→  label: string;
   722→  value: number | null;
   723→  expectedValue: number | null;
   724→  format: 'currency' | 'multiplier' | 'payback';
   725→  showDelta?: boolean;
   726→}) {
   727→  // Delta vs. expected — only rendered when showDelta (active != 'expected') AND both
   728→  // values are non-null.
   729→  const delta =
   730→    showDelta && value != null && expectedValue != null
   731→      ? formatDelta(value, expectedValue, format)
   732→      : null;
   733→  return (
   734→    <div className="mkt-card-quiet p-5">
   735→      <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-ink-muted">{label}</p>
   736→      <p className="mt-3 font-mono tnum text-[clamp(1.75rem,3vw,2.25rem)] font-bold tracking-[-0.02em] text-ink">
   737→        {format === 'currency' ? (
   738→          <CountUp
   739→            value={value ?? 0}
   740→            prefix="$"
   741→            decimals={0}
   742→            retriggerOnValueChange
   743→          />
   744→        ) : format === 'multiplier' ? (
   745→          value == null ? (
   746→            <span>N/A</span>
   747→          ) : (
   748→            <CountUp
   749→              value={value}
   750→              suffix="×"
   751→              decimals={1}
   752→              retriggerOnValueChange
   753→            />
   754→          )
   755→        ) : value == null ? (
   756→          <span>Never</span>
   757→        ) : (
   758→          <CountUp
   759→            value={value}
   760→            suffix=" months"
   761→            decimals={1}
   762→            retriggerOnValueChange
   763→          />
   764→        )}
   765→      </p>
   766→      {delta && (
   767→        <p className="mt-2 font-mono tnum text-[11px] text-ink-faint">{delta}</p>
   768→      )}
   769→    </div>
   770→  );
   771→}
   772→
   773→/** Format a scenario delta vs. expected: "+$58k vs expected" / "−3.9×" / "+5.5 mo". */
   774→function formatDelta(
   775→  value: number,
   776→  expectedValue: number,
   777→  format: 'currency' | 'multiplier' | 'payback'
   778→): string {
   779→  const diff = value - expectedValue;
   780→  const sign = diff > 0 ? '+' : diff < 0 ? '\u2212' : '';
   781→  const abs = Math.abs(diff);
   782→  if (format === 'currency') {
   783→    // Compact k tier for deltas — "$58k" not "$58,000".
   784→    const v =
   785→      abs >= 1000
   786→        ? `${(abs / 1000).toFixed(abs >= 10000 ? 0 : 1)}k`
   787→        : `${abs.toFixed(0)}`;
   788→    return `${sign}$${v} vs expected`;
   789→  }
   790→  if (format === 'multiplier') {
   791→    return `${sign}${abs.toFixed(1)}× vs expected`;
   792→  }
   793→  // payback
   794→  return `${sign}${abs.toFixed(1)} mo vs expected`;
   795→}
