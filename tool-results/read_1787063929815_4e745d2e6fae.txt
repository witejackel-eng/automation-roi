    63→export function ResultsView() {
    64→  const {
    65→    inputs,
    66→    results,
    67→    recommendation,
    68→    entitlement,
    69→    go,
    70→    activeScenario,
    71→    setActiveScenario,
    72→  } = useApp(
    73→    useShallow((s) => ({
    74→      inputs: s.inputs,
    75→      results: s.results,
    76→      recommendation: s.recommendation,
    77→      entitlement: s.entitlement,
    78→      go: s.go,
    79→      activeScenario: s.activeScenario,
    80→      setActiveScenario: s.setActiveScenario,
    81→    }))
    82→  );
    83→
    84→  const { toast } = useToast();
    85→  const [saving, setSaving] = React.useState(false);
    86→  const [reportLoading, setReportLoading] = React.useState(false);
    87→  const [proposalLoading, setProposalLoading] = React.useState(false);
    88→  const [shareLoading, setShareLoading] = React.useState(false);
    89→  // Mirror the store's savedProjectId locally so the existing handlers keep
    90→  // working unchanged, but ALSO sync it back to the store on save so that
    91→  // the reopen flow (which sets the store's savedProjectId before navigating
    92→  // here) is respected — re-saving after a reopen updates the same project
    93→  // rather than creating a duplicate.
    94→  const storeSavedProjectId = useApp((s) => s.savedProjectId);
    95→  const [savedProjectId, setSavedProjectId] = React.useState<string | null>(
    96→    storeSavedProjectId
    97→  );
    98→  // If the store's savedProjectId changes (e.g. user reopened a project),
    99→  // sync the local state.
   100→  React.useEffect(() => {
   101→    setSavedProjectId(storeSavedProjectId);
   102→  }, [storeSavedProjectId]);
   103→
   104→  if (!inputs || !results || !recommendation) {
   105→    return (
   106→      <div className="mx-auto w-full max-w-[1200px] px-4 py-16 md:px-6">
   107→        <EmptyResults onBack={() => go('calculator')} />
   108→      </div>
   109→    );
   110→  }
   111→
   112→  // Active scenario drives every displayed figure. The store's
   113→  // `recommendation` (Expected) is preserved for the saved-project record.
   114→  const active: ScenarioResult = results[activeScenario];
   115→  const activeRec = recommend(active);
   116→  const activeDecision = activeRec.recommendation as DecisionKey;
   117→  const ent = entitlement;
   118→  const canSave = !!ent && has(ent, 'save_project');
   119→  const canReport = !!ent && has(ent, 'client_report');
   120→  const canProposal = !!ent && has(ent, 'proposal');
   121→  const canShare = !!ent && has(ent, 'share_links');
   122→
   123→  // ROI as a multiplier (499% → "5.0×"). Null when totalFirstYearCost = 0.
   124→  const roiMultiple =
   125→    active.roiPct == null ? null : active.roiPct / 100;
   126→
   127→  const handleSave = async () => {
   128→    if (!canSave) {
   129→      go('pricing');
   130→      return;
   131→    }
   132→    setSaving(true);
   133→    try {
   134→      const res = await fetch('/api/projects', {
   135→        method: 'POST',
   136→        headers: { 'Content-Type': 'application/json' },
   137→        body: JSON.stringify({
   138→          inputs,
   139→          results,
   140→          recommendation,
   141→          clientName: inputs.clientName,
   142→        }),
   143→      });
   144→      if (res.status === 403) {
   145→        toast({ title: 'Saving projects requires Pro.', variant: 'destructive' });
   146→        go('pricing');
   147→        return;
   148→      }
   149→      if (!res.ok) {
   150→        toast({ title: 'Could not save the project.', variant: 'destructive' });
   151→        return;
   152→      }
   153→      const data = (await res.json()) as { id: string };
   154→      setSavedProjectId(data.id);
   155→      toast({ title: 'Project saved.' });
   156→    } catch {
   157→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   158→    } finally {
   159→      setSaving(false);
   160→    }
   161→  };
   162→
   163→  const handleReport = async () => {
   164→    if (!canReport) {
   165→      go('pricing');
   166→      return;
   167→    }
   168→    if (!savedProjectId) {
   169→      await handleSave();
   170→    }
   171→    const projectId = savedProjectId;
   172→    if (!projectId) {
   173→      toast({ title: 'Save the project first.', description: 'Then generate the report.' });
   174→      return;
   175→    }
   176→    setReportLoading(true);
   177→    try {
   178→      const res = await fetch(`/api/projects/${projectId}/report`, {
   179→        method: 'POST',
   180→        headers: { 'Content-Type': 'application/json' },
   181→        body: JSON.stringify({}),
   182→      });
   183→      if (res.status === 403) {
   184→        toast({ title: 'Client report requires Pro.', variant: 'destructive' });
   185→        go('pricing');
   186→        return;
   187→      }
   188→      if (!res.ok) {
   189→        toast({ title: 'Could not generate the report.', variant: 'destructive' });
   190→        return;
   191→      }
   192→      const data = (await res.json()) as { pdfUrl: string };
   193→      toast({ title: 'Report generated.' });
   194→      window.open(data.pdfUrl, '_blank', 'noopener,noreferrer');
   195→    } catch {
   196→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   197→    } finally {
   198→      setReportLoading(false);
   199→    }
   200→  };
   201→
   202→  const handleProposal = async () => {
   203→    if (!canProposal) {
   204→      go('pricing');
   205→      return;
   206→    }
   207→    if (!savedProjectId) {
   208→      await handleSave();
   209→    }
   210→    const projectId = savedProjectId;
   211→    if (!projectId) {
   212→      toast({ title: 'Save the project first.', description: 'Then generate the proposal.' });
   213→      return;
   214→    }
   215→    setProposalLoading(true);
   216→    try {
   217→      const res = await fetch(`/api/projects/${projectId}/proposal`, {
   218→        method: 'POST',
   219→        headers: { 'Content-Type': 'application/json' },
   220→        body: JSON.stringify({}),
   221→      });
   222→      if (res.status === 403) {
   223→        toast({ title: 'Proposal generation requires Pro.', variant: 'destructive' });
   224→        go('pricing');
   225→        return;
   226→      }
   227→      if (!res.ok) {
   228→        toast({ title: 'Could not generate the proposal.', variant: 'destructive' });
   229→        return;
   230→      }
   231→      const data = (await res.json()) as { pdfUrl: string };
   232→      toast({ title: 'Proposal generated.' });
   233→      window.open(data.pdfUrl, '_blank', 'noopener,noreferrer');
   234→    } catch {
   235→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   236→    } finally {
   237→      setProposalLoading(false);
   238→    }
   239→  };
   240→
   241→  const handleShare = async () => {
   242→    if (!canShare) {
   243→      go('pricing');
   244→      return;
   245→    }
   246→    if (!savedProjectId) {
   247→      await handleSave();
   248→    }
   249→    const projectId = savedProjectId;
   250→    if (!projectId) {
   251→      toast({ title: 'Save the project first.', description: 'Then create a share link.' });
   252→      return;
   253→    }
   254→    setShareLoading(true);
   255→    try {
   256→      const res = await fetch(`/api/projects/${projectId}/share`, {
   257→        method: 'POST',
   258→        headers: { 'Content-Type': 'application/json' },
   259→        body: JSON.stringify({}),
   260→      });
   261→      if (res.status === 403) {
   262→        toast({ title: 'Share links require Agency Pro.', variant: 'destructive' });
   263→        go('pricing');
   264→        return;
   265→      }
   266→      if (!res.ok) {
   267→        toast({ title: 'Could not create the share link.', variant: 'destructive' });
   268→        return;
   269→      }
   270→      const data = (await res.json()) as { shareId: string; url: string };
   271→      // Copy to clipboard + use Web Share API where supported (Master Spec §46).
   272→      const fullUrl = `${window.location.origin}${data.url}`;
   273→      if (navigator.share) {
   274→        try {
   275→          await navigator.share({
   276→            title: 'Viableo — Shared analysis',
   277→            text: 'A shared automation business case.',
   278→            url: fullUrl,
   279→          });
   280→        } catch {
   281→          /* user dismissed — still copy below */
   282→        }
   283→      }
   284→      try {
   285→        await navigator.clipboard.writeText(fullUrl);
   286→        toast({
   287→          title: 'Link copied.',
   288→          description: 'Share it with your client.',
   289→        });
   290→      } catch {
   291→        toast({
   292→          title: 'Share link created.',
   293→          description: fullUrl,
   294→        });
   295→      }
   296→    } catch {
   297→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   298→    } finally {
   299→      setShareLoading(false);
   300→    }
   301→  };
   302→
   303→  return (
   304→    <div className="mx-auto w-full max-w-[1200px] px-4 py-8 md:px-6 md:py-12">
   305→      <ScrollToTop />
   306→      {/* Back + prepared-for row */}
   307→      <div className="mb-6 flex items-center justify-between">
   308→        <Button
   309→          variant="ghost"
   310→          size="sm"
   311→          onClick={() => go('calculator')}
   312→          className="min-h-[44px] gap-1.5 text-ink-muted hover:text-ink"
   313→        >
   314→          <ArrowLeft className="size-4" strokeWidth={1.75} aria-hidden="true" />
   315→          Edit assumptions
   316→        </Button>
   317→        <p className="text-[13px] text-ink-muted">
   318→          Prepared for <span className="font-medium text-ink">{inputs.clientName}</span>
   319→        </p>
   320→      </div>
   321→
   322→      {/* Page header — references TERM.analysis + TERM.decision */}
   323→      <header className="mb-8">
   324→        <h1 className="font-display text-[32px] font-bold leading-[1.02] tracking-[-0.02em] text-ink md:text-[40px]">
   325→          {TERM.analysis}
   326→        </h1>
   327→        <p className="mt-2 max-w-[640px] text-[15px] leading-[1.55] text-ink-muted">

   329→          numbers hold up under conservative and upside assumptions.
   330→        </p>
   331→      </header>
   332→
   333→      {/* Hero summary card — ScenarioSlider + DecisionBadge + 3 headline figures */}
   334→      <section
   335→        aria-label="Viableo Decision summary"
   336→        className="mb-10 rounded-lg border border-border bg-surface-raised p-5 md:p-6"

   339→          <ScenarioSlider value={activeScenario} onChange={setActiveScenario} />
   340→          {/* Cross-fade on decision change via key + reveal-on-enter animation. */}
   341→          <div
   342→            key={activeDecision}
   343→            className="reveal-on-enter flex items-center gap-2"
   344→          >
   345→            <span className="text-[11px] uppercase tracking-wide text-ink-muted">
   346→              {TERM.decision}
   347→            </span>
   348→            <DecisionBadge decision={activeDecision} size="lg" animate />
   349→          </div>
   350→        </div>
   351→
   352→        <AnimatePresence mode="wait">
   353→          <motion.div
   354→            key={activeScenario}
   355→            initial={{ opacity: 0.6, y: 4 }}
   356→            animate={{ opacity: 1, y: 0 }}
   357→            exit={{ opacity: 0.6, y: -4 }}
   358→            transition={{ duration: 0.28, ease: [0.16, 1, 0.3, 1] }}
   359→            className="grid grid-cols-1 gap-6 sm:grid-cols-3 sm:gap-8"
   360→          >
   361→          {/* Annual Opportunity ($) */}
   362→          <HeroFigure
   363→            label="Annual opportunity"
   364→            sublabel={
   365→              active.isRevenueOpportunityOnly
   366→                ? 'Revenue opportunity, net of first-year cost'
   367→                : 'Net annual benefit, net of first-year cost'
   368→            }
   369→          >
   370→            <CountUp
   371→              value={active.netAnnualBenefit}
   372→              prefix="$"
   373→              decimals={0}
   374→              retriggerOnValueChange
   375→              className="text-[40px] font-medium leading-none text-ink"
   376→            />
   377→          </HeroFigure>
   378→
   379→          {/* ROI (×) */}
   380→          <HeroFigure label="ROI" sublabel="Return on first-year cost">
   381→            {roiMultiple == null ? (
   382→              <span className="font-mono tnum text-[40px] font-medium leading-none text-ink-muted">
   383→                N/A
   384→              </span>
   385→            ) : (
   386→              <CountUp
   387→                value={roiMultiple}
   388→                suffix="×"
   389→                decimals={1}
   390→                retriggerOnValueChange
   391→                className="text-[40px] font-medium leading-none text-ink"
   392→              />
   393→            )}
   394→          </HeroFigure>
   395→
   396→          {/* Payback (months) */}
   397→          <HeroFigure label="Payback" sublabel="Months to recoup first-year cost">

   399→              <span className="font-mono tnum text-[40px] font-medium leading-none text-ink-muted">
   400→                Never
   401→              </span>
   402→            ) : (
   403→              <CountUp
   404→                value={active.paybackMonths}
   405→                suffix=" mo"
   406→                decimals={active.paybackMonths >= 10 ? 0 : 1}
   407→                retriggerOnValueChange
   408→                className="text-[40px] font-medium leading-none text-ink"
   409→              />
   410→            )}
   411→          </HeroFigure>
   412→        </motion.div>
   413→        </AnimatePresence>
   414→      </section>
   415→
   416→      {/* Verdict Stamp — the detailed sign-off (kept per Task 4-b). */}
   417→      <div className="mb-8 flex justify-center md:justify-start">
   418→        <VerdictStamp
   419→          recommendation={activeRec.recommendation}
   420→          paybackMonths={active.paybackMonths}
   421→          roiPct={active.roiPct}
   422→          netAnnualBenefit={active.netAnnualBenefit}
   423→          size="lg"
   424→        />
   425→      </div>
   426→
   427→      {/* Why this recommendation? (Section 6.9) — expandable bullet rationale,
   428→          each reason backed by a real number from the active scenario. */}
   429→      <WhyRecommendation
   430→        inputs={inputs}
   431→        results={results}
   432→        recommendation={activeRec}
   433→        className="mb-8"
   434→      />
   435→
   436→      {/* Recommendation copy (active scenario) */}
   437→      <p className="mb-10 max-w-[760px] text-[15px] leading-[1.55] text-ink-muted">
   438→        {activeRec.copy}
   439→      </p>
   440→
   441→      {/* Secondary KPI row */}
   442→      <div className="mb-10 grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   443→        <KpiCard
   444→          label="Annual labor savings"
   445→          value={formatCurrency(active.annualLaborSavings)}
   446→          variant="secondary"
   447→        />
   448→        <KpiCard
   449→          label={
   450→            active.isRevenueOpportunityOnly
   451→              ? 'Additional revenue opportunity'
   452→              : 'Additional gross profit'
   453→          }
   454→          value={formatCurrency(active.additionalGrossProfit)}
   455→          variant="secondary"
   456→        />
   457→        <KpiCard
   458→          label="First-year cost"
   459→          value={formatCurrency(active.totalFirstYearCost)}
   460→          variant="secondary"
   461→        />
   462→      </div>
   463→
   464→      {/* Stress test + break-even thresholds (Master Spec §29, §30, §31).
   465→          Placed after the key financials and before the exhibits, per the
   466→          §33 results-page section order: decision → financials → why →
   467→          confidence → stress test → break-even → ROI bridge → comparison →

   469→      <StressTestSection inputs={inputs} activeScenario={activeScenario} />
   470→
   471→      {/* Top drivers callout (Phase 2.4) */}
   472→      <TopDriversCallout inputs={inputs} activeScenario={activeScenario} />
   473→
   474→      {/* Exhibit 1 — ROI Bridge (active scenario) */}
   475→      <React.Suspense
   476→        fallback={
   477→          <ChartSkeleton title="Exhibit 1 — Labor savings drive the first-year benefit" />
   478→        }
   479→      >
   480→        <div className="mb-10">
   481→          <RoiBridge
   482→            result={active}
   483→            isRevenueOpportunityOnly={active.isRevenueOpportunityOnly}
   484→          />
   485→        </div>
   486→      </React.Suspense>
   487→
   488→      {/* Exhibit 2 — Scenario Comparison (all three) */}
   489→      <React.Suspense
   490→        fallback={
   491→          <ChartSkeleton title="Exhibit 2 — Expected scenario carries the strongest benefit" />
   492→        }
   493→      >
   494→        <div className="mb-10">
   495→          <ScenarioComparison
   496→            results={results}
   497→            isRevenueOpportunityOnly={active.isRevenueOpportunityOnly}
   498→          />
   499→        </div>
   500→      </React.Suspense>
   501→
   502→      {/* Scenario detail table (numeric, for the audit trail) */}
   503→      <ScenarioTable
   504→        results={results}
   505→        activeScenario={activeScenario}
   506→        onSelect={setActiveScenario}
   507→      />
   508→
   509→      {/* Assumptions */}
   510→      <div className="mb-10 mt-12">
   511→        <h2 className="mb-4 font-display text-[22px] font-semibold tracking-[-0.02em] text-ink">
   512→          Assumptions
   513→        </h2>
   514→        <p className="mb-4 text-[14px] text-ink-muted">
   515→          Every input that feeds the calculation. Fields tagged &quot;context&quot; do not

   517→        </p>
   518→        <AssumptionsTable inputs={inputs} />
   519→      </div>
   520→
   521→      {/* Sticky secondary actions */}
   522→      <div className="sticky bottom-0 -mx-4 mt-8 border-t border-border bg-canvas/95 px-4 py-4 backdrop-blur supports-[backdrop-filter]:bg-canvas/80 md:static md:mx-0 md:bg-transparent md:px-0 md:py-0 md:backdrop-blur-none">

   525→            allowed={canSave}
   526→            requiredTierLabel="Pro"
   527→            onClick={handleSave}
   528→            loading={saving}
   529→            variant="secondary"
   530→            icon={<Save className="size-4" strokeWidth={1.75} aria-hidden="true" />}
   531→          >
   532→            Save project
   533→          </EntitlementButton>
   534→          <EntitlementButton
   535→            allowed={canReport}
   536→            requiredTierLabel="Pro"
   537→            onClick={handleReport}
   538→            loading={reportLoading}
   539→            icon={<FileText className="size-4" strokeWidth={1.75} aria-hidden="true" />}
   540→          >
   541→            Generate Client Report
   542→          </EntitlementButton>
   543→          <EntitlementButton
   544→            allowed={canProposal}
   545→            requiredTierLabel="Pro"
   546→            onClick={handleProposal}
   547→            loading={proposalLoading}
   548→            icon={<FileSignature className="size-4" strokeWidth={1.75} aria-hidden="true" />}
   549→          >

   551→          </EntitlementButton>
   552→          <EntitlementButton
   553→            allowed={canShare}
   554→            requiredTierLabel="Agency Pro"
   555→            onClick={handleShare}
   556→            loading={shareLoading}
   557→            icon={<Share2 className="size-4" strokeWidth={1.75} aria-hidden="true" />}
   558→          >
   559→            Create share link
   560→          </EntitlementButton>
   561→        </div>
   562→      </div>
   563→    </div>
   564→  );

   569→function HeroFigure({
   570→  label,
   571→  sublabel,
   572→  children,
   573→}: {
   574→  label: string;

   719→/** Floating scroll-to-top button — appears after 400px scroll. */
   720→function ScrollToTop() {
   721→  const [visible, setVisible] = React.useState(false);
   722→
   723→  React.useEffect(() => {
   724→    const onScroll = () => setVisible(window.scrollY > 400);
   725→    onScroll();
   726→    window.addEventListener('scroll', onScroll, { passive: true });
   727→    return () => window.removeEventListener('scroll', onScroll);
   728→  }, []);
   729→
   730→  if (!visible) return null;
   731→
   732→  return (
   733→    <button
   734→      type="button"
   735→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   736→      className="fixed bottom-20 right-4 z-20 flex size-10 items-center justify-center rounded-full border border-border bg-surface shadow-floating text-ink-muted hover:text-ink transition-colors duration-hover md:bottom-6"
   737→      aria-label="Back to top"
   738→    >
   739→      <ArrowUp className="size-4" strokeWidth={1.75} aria-hidden="true" />
   740→    </button>
   741→  );
   742→}
