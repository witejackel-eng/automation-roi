   705→   705→/** Floating scroll-to-top button — appears after 400px scroll. */
   706→   706→function ScrollToTop() {
   707→   707→  const [visible, setVisible] = React.useState(false);
   708→   708→
   709→   709→  React.useEffect(() => {
   710→   710→    const onScroll = () => setVisible(window.scrollY > 400);
   711→   711→    onScroll();
   712→   712→    window.addEventListener('scroll', onScroll, { passive: true });
   713→   713→    return () => window.removeEventListener('scroll', onScroll);
   714→   714→  }, []);
   715→   715→
   716→   716→  if (!visible) return null;
   717→   717→
   718→   718→  return (
   719→   719→    <button
   720→   720→      type="button"
   721→   721→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   722→   722→      className="fixed bottom-20 right-4 z-20 flex size-10 items-center justify-center rounded-full border border-border bg-surface shadow-floating text-ink-muted hover:text-ink transition-colors duration-hover md:bottom-6"
   723→   723→      aria-label="Back to top"
   724→   724→    >
   725→   725→      <ArrowUp className="size-4" strokeWidth={1.75} aria-hidden="true" />
   726→   726→    </button>
   727→   727→  );
   728→   728→}
