    86→/** Render an ROI% as a multiplier (499% → "5.0×"). */
    87→function roiAsMultiplier(roiPct: number | null | undefined): string {
    88→  if (roiPct == null || !Number.isFinite(roiPct)) return 'N/A';
    89→  return `${(roiPct / 100).toFixed(1)}×`;

   679→function ScenarioStat({
   680→  label,
   681→  value,
   682→  expectedValue,
   683→  format,
   684→  showDelta,
   685→}: {
   686→  label: string;
   687→  value: number | null;
   688→  expectedValue: number | null;
   689→  format: 'currency' | 'multiplier' | 'payback';
   690→  showDelta?: boolean;
   691→}) {
   692→  // Delta vs. expected — only rendered when showDelta (active != 'expected') AND both
   693→  // values are non-null.
   694→  const delta =
   695→    showDelta && value != null && expectedValue != null
   696→      ? formatDelta(value, expectedValue, format)
   697→      : null;
   698→  return (
   699→    <div className="mkt-card-quiet p-5">
   700→      <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-ink-muted">{label}</p>
   701→      <p className="mt-3 font-mono tnum text-[clamp(1.75rem,3vw,2.25rem)] font-bold tracking-[-0.02em] text-ink">
   702→        {format === 'currency' ? (
   703→          <CountUp
   704→            value={value ?? 0}
   705→            prefix="$"
   706→            decimals={0}
   707→            retriggerOnValueChange
   708→          />
   709→        ) : format === 'multiplier' ? (
   710→          value == null ? (
   711→            <span>N/A</span>
   712→          ) : (
   713→            <CountUp
   714→              value={value}
   715→              suffix="×"
   716→              decimals={1}
   717→              retriggerOnValueChange
   718→            />
   719→          )
   720→        ) : value == null ? (
   721→          <span>Never</span>
   722→        ) : (
   723→          <CountUp
   724→            value={value}
   725→            suffix=" months"
   726→            decimals={1}
   727→            retriggerOnValueChange
   728→          />
   729→        )}
   730→      </p>
   731→      {delta && (
   732→        <p className="mt-2 font-mono tnum text-[11px] text-ink-faint">{delta}</p>
   733→      )}
   734→    </div>
   735→  );
   736→}

   738→/** Format a scenario delta vs. expected: "+$58k vs expected" / "−3.9×" / "+5.5 mo". */
   739→function formatDelta(
   740→  value: number,
   741→  expectedValue: number,
   742→  format: 'currency' | 'multiplier' | 'payback'
   743→): string {
   744→  const diff = value - expectedValue;
   745→  const sign = diff > 0 ? '+' : diff < 0 ? '\u2212' : '';
   746→  const abs = Math.abs(diff);
   747→  if (format === 'currency') {
   748→    // Compact k tier for deltas — "$58k" not "$58,000".
   749→    const v =
   750→      abs >= 1000
   751→        ? `${(abs / 1000).toFixed(abs >= 10000 ? 0 : 1)}k`
   752→        : `${abs.toFixed(0)}`;
   753→    return `${sign}$${v} vs expected`;
   754→  }
   755→  if (format === 'multiplier') {
   756→    return `${sign}${abs.toFixed(1)}× vs expected`;
   757→  }
   758→  // payback
   759→  return `${sign}${abs.toFixed(1)} mo vs expected`;
   760→}
