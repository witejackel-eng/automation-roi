   121→   121→  // ROI as a multiplier (499% → "5.0×"). Null when totalFirstYearCost = 0.

   123→   123→    active.roiPct == null ? null : active.roiPct / 100;

   158→   158→    }

   163→   163→      go('pricing');

   167→   167→      await handleSave();

   180→   180→      });

   182→   182→        toast({ title: 'Client report requires Pro.', variant: 'destructive' });
   183→   183→        go('pricing');
   184→   184→        return;
   185→   185→      }
   186→   186→      if (!res.ok) {
   187→   187→        toast({ title: 'Could not generate the report.', variant: 'destructive' });

   190→   190→      const data = (await res.json()) as { pdfUrl: string };
   191→   191→      toast({ title: 'Report generated.' });

   193→   193→    } catch {
   194→   194→      toast({ title: 'Could not reach the service.', variant: 'destructive' });
   195→   195→    } finally {
   196→   196→      setReportLoading(false);
   197→   197→    }

   199→   199→

   201→   201→    if (!canProposal) {
   202→   202→      go('pricing');
   203→   203→      return;
   204→   204→    }
   205→   205→    if (!savedProjectId) {
   206→   206→      await handleSave();
   207→   207→    }
   208→   208→    const projectId = savedProjectId;

   233→   233→      toast({ title: 'Could not reach the service.', variant: 'destructive' });

   235→   235→      setProposalLoading(false);
   236→   236→    }
   237→   237→  };

   241→   241→      go('pricing');
   242→   242→      return;
   243→   243→    }
   244→   244→    if (!savedProjectId) {
   245→   245→      await handleSave();

   248→   248→    if (!projectId) {
   249→   249→      toast({ title: 'Save the project first.', description: 'Then create a share link.' });
   250→   250→      return;
   251→   251→    }
   252→   252→    setShareLoading(true);
   253→   253→    try {
   254→   254→      const res = await fetch(`/api/projects/${projectId}/share`, {
   255→   255→        method: 'POST',
   256→   256→        headers: { 'Content-Type': 'application/json' },
   257→   257→        body: JSON.stringify({}),
   258→   258→      });

   328→   328→        </p>
   329→   329→      </header>
   330→   330→

   338→   338→          {/* Cross-fade on decision change via key + reveal-on-enter animation. */}

   341→   341→            className="reveal-on-enter flex items-center gap-2"

   350→   350→        <AnimatePresence mode="wait">

   422→   422→        />

   424→   424→
   425→   425→      {/* Why this recommendation? (Section 6.9) — expandable bullet rationale,

   433→   433→
   434→   434→      {/* Recommendation copy (active scenario) */}

   468→   468→

   714→   714→/** Floating scroll-to-top button — appears after 400px scroll. */
   715→   715→function ScrollToTop() {
   716→   716→  const [visible, setVisible] = React.useState(false);
   717→   717→
   718→   718→  React.useEffect(() => {
   719→   719→    const onScroll = () => setVisible(window.scrollY > 400);
   720→   720→    onScroll();
   721→   721→    window.addEventListener('scroll', onScroll, { passive: true });
   722→   722→    return () => window.removeEventListener('scroll', onScroll);
   723→   723→  }, []);
   724→   724→
   725→   725→  if (!visible) return null;
   726→   726→
   727→   727→  return (
   728→   728→    <button
   729→   729→      type="button"
   730→   730→      onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   731→   731→      className="fixed bottom-20 right-4 z-20 flex size-10 items-center justify-center rounded-full border border-border bg-surface shadow-floating text-ink-muted hover:text-ink transition-colors duration-hover md:bottom-6"
   732→   732→      aria-label="Back to top"
   733→   733→    >
   734→   734→      <ArrowUp className="size-4" strokeWidth={1.75} aria-hidden="true" />
   735→   735→    </button>
   736→   736→  );
   737→   737→}
