   714→   714→function ScenarioStat({
   715→   715→  label,
   716→   716→  value,
   717→   717→  expectedValue,
   718→   718→  format,
   719→   719→  showDelta,
   720→   720→}: {
   721→   721→  label: string;
   722→   722→  value: number | null;
   723→   723→  expectedValue: number | null;
   724→   724→  format: 'currency' | 'multiplier' | 'payback';
   725→   725→  showDelta?: boolean;
   726→   726→}) {
   727→   727→  // Delta vs. expected — only rendered when showDelta (active != 'expected') AND both
   728→   728→  // values are non-null.
   729→   729→  const delta =
   730→   730→    showDelta && value != null && expectedValue != null
   731→   731→      ? formatDelta(value, expectedValue, format)
   732→   732→      : null;
   733→   733→  return (
   734→   734→    <div className="mkt-card-quiet p-5">
   735→   735→      <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-ink-muted">{label}</p>
   736→   736→      <p className="mt-3 font-mono tnum text-[clamp(1.75rem,3vw,2.25rem)] font-bold tracking-[-0.02em] text-ink">
   737→   737→        {format === 'currency' ? (
   738→   738→          <CountUp
   739→   739→            value={value ?? 0}
   740→   740→            prefix="$"
   741→   741→            decimals={0}
   742→   742→            retriggerOnValueChange
   743→   743→          />
   744→   744→        ) : format === 'multiplier' ? (
   745→   745→          value == null ? (
   746→   746→            <span>N/A</span>
   747→   747→          ) : (
   748→   748→            <CountUp
   749→   749→              value={value}
   750→   750→              suffix="×"
   751→   751→              decimals={1}
   752→   752→              retriggerOnValueChange
   753→   753→            />
   754→   754→          )
   755→   755→        ) : value == null ? (
   756→   756→          <span>Never</span>
   757→   757→        ) : (
   758→   758→          <CountUp
   759→   759→            value={value}
   760→   760→            suffix=" months"
   761→   761→            decimals={1}
   762→   762→            retriggerOnValueChange
   763→   763→          />
   764→   764→        )}
   765→   765→      </p>
   766→   766→      {delta && (
   767→   767→        <p className="mt-2 font-mono tnum text-[11px] text-ink-faint">{delta}</p>
   768→   768→      )}
   769→   769→    </div>
   770→   770→  );
   771→   771→}
   772→   772→
   773→   773→/** Format a scenario delta vs. expected: "+$58k vs expected" / "−3.9×" / "+5.5 mo". */
   774→   774→function formatDelta(
   775→   775→  value: number,
   776→   776→  expectedValue: number,
   777→   777→  format: 'currency' | 'multiplier' | 'payback'
   778→   778→): string {
   779→   779→  const diff = value - expectedValue;
   780→   780→  const sign = diff > 0 ? '+' : diff < 0 ? '\u2212' : '';
   781→   781→  const abs = Math.abs(diff);
   782→   782→  if (format === 'currency') {
   783→   783→    // Compact k tier for deltas — "$58k" not "$58,000".
   784→   784→    const v =
   785→   785→      abs >= 1000
   786→   786→        ? `${(abs / 1000).toFixed(abs >= 10000 ? 0 : 1)}k`
   787→   787→        : `${abs.toFixed(0)}`;
   788→   788→    return `${sign}$${v} vs expected`;
   789→   789→  }
   790→   790→  if (format === 'multiplier') {
   791→   791→    return `${sign}${abs.toFixed(1)}× vs expected`;
   792→   792→  }
   793→   793→  // payback
   794→   794→  return `${sign}${abs.toFixed(1)} mo vs expected`;
   795→   795→}
